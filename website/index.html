One Engine. Every Language.

Write your bot in the syntax you love without sacrificing native performance.

main.rs 
// Rust (main.rs)
use rift::prelude::*;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let client = Client::builder(
        std::env::var("DISCORD_TOKEN")?,
        Intents::GUILDS | Intents::GUILD_MESSAGES,
    )
    .build()
    .await?;

    let router = CommandRouter::new()
        .slash("ping", "Check bot latency", |ctx| async move {
            ctx.reply("Pong from Rift! ⚡").await?;
            Ok(())
        });

    let mut framework = Framework::new(client, router);
    framework.run().await?;
    Ok(())
}
